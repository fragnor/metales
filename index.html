/** * METALES APP - BACKEND V4.9 */ function doPost(e) { var data = e.parameter; var accion = data.accion; try { switch(accion) { case 'login': return respuesta(hacerLogin(data.user, data.pass)); case 'getProduccion': return respuesta(obtenerDatosCompletos(data)); case 'guardarHoras': return respuesta(guardarHoras(data)); case 'guardarMD': return respuesta(guardarMediosDensos(data)); case 'guardarIncidencia': return respuesta(guardarIncidencia(data)); case 'borrarIncidencia': return respuesta(borrarIncidencia(data)); case 'guardarPrueba': return respuesta(guardarPrueba(data)); case 'borrarProduccion': return respuesta(borrarProduccion(data)); default: return respuesta({ success: false, error: "Acción no reconocida" }); } } catch (err) { return respuesta({ success: false, error: err.toString() }); } } function respuesta(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function obtenerDatosCompletos(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var tz = ss.getSpreadsheetTimeZone(); var res = { success: true, actual: [], anterior: [], hMaq: 0, hProd: 0, cont: 0, horas: 0, ultimasInc: [] }; var fActualStr = String(data.fecha).trim(); var tActual = String(data.turno).trim(); var parts = fActualStr.split("-"); var dPrev = new Date(parts[0], parts[1] - 1, parts[2], 12, 0, 0); var tPrev = (tActual === "Mañana") ? "Noche" : (tActual === "Tarde" ? "Mañana" : "Tarde"); if (tActual === "Mañana") dPrev.setDate(dPrev.getDate() - 1); var fPrevStr = Utilities.formatDate(dPrev, tz, "yyyy-MM-dd"); var sheetMaq = ss.getSheetByName(data.maquina); if (sheetMaq) { var vals = sheetMaq.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { var cellVal = vals[i][0]; var fFila = (cellVal instanceof Date) ? Utilities.formatDate(cellVal, tz, "yyyy-MM-dd") : String(cellVal).trim().substring(0, 10); var tFila = String(vals[i][1]).trim(); var item = {}; if (data.maquina === "Medios densos") { item = { id: String(vals[i][16]), lote: vals[i][14], hMaq: vals[i][5], usuario: vals[i][2], hMaqIni: vals[i][3], hMaqFin: vals[i][4], hProdIni: vals[i][6], hProdFin: vals[i][7], contIni: vals[i][9], contFin: vals[i][10], fesi: vals[i][12], obs: vals[i][13], m3Agua: vals[i][15] }; } else { item = { id: String(vals[i][8]), lote: vals[i][7], hIni: vals[i][4], hFin: vals[i][5] }; } if (fFila === fActualStr && tFila === tActual) res.actual.push(item); if (fFila === fPrevStr && tFila === tPrev) res.anterior.push(item); } if (vals.length > 1) { var last = vals[vals.length - 1]; if (data.maquina === 'Medios densos') { res.hMaq = last[4]; res.hProd = last[7]; res.cont = last[10]; } else { res.horas = last[5]; } } } var sheetInc = ss.getSheetByName("incidencias " + data.maquina); if (sheetInc && sheetInc.getLastRow() > 1) { var dataInc = sheetInc.getRange(Math.max(2, sheetInc.getLastRow()-2), 1, Math.min(3, sheetInc.getLastRow()-1), 10).getValues(); res.ultimasInc = dataInc.reverse().map(function(f) { return { motivo: f[6], activo: f[7], obs: f[8], horas: f[9] ? parseFloat(f[9]).toFixed(2) : "0" }; }); } return res; } function guardarHoras(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(data.maquina); var hTot = (parseFloat(data.horasFin) || 0) - (parseFloat(data.horasIni) || 0); var id = (data.id && String(data.id).trim() !== "") ? String(data.id).trim() : Utilities.getUuid(); var record = { id: id, lote: data.lote, hIni: data.horasIni, hFin: data.horasFin }; if (data.id && String(data.id).trim() !== "") { var vals = sheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][8]) === id) { sheet.getRange(i+1, 1, 1, 9).setValues([[data.fecha, data.turno, data.usuario, new Date(), data.horasIni, data.horasFin, hTot, data.lote, id]]); return { success: true, record: record }; } } } sheet.appendRow([data.fecha, data.turno, data.usuario, new Date(), data.horasIni, data.horasFin, hTot, data.lote, id]); return { success: true, record: record }; } function guardarMediosDensos(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Medios densos"); var id = (data.id && String(data.id).trim() !== "") ? String(data.id).trim() : Utilities.getUuid(); var dMaq = (parseFloat(data.hMaqFin) || 0) - (parseFloat(data.hMaqIni) || 0); var dProd = (parseFloat(data.hProdFin) || 0) - (parseFloat(data.hProdIni) || 0); var dCont = (parseFloat(data.contFin) || 0) - (parseFloat(data.contIni) || 0); // Columna P (indice 15) = Agua, Columna Q (indice 16) = ID var fila = [ data.fecha, data.turno, data.operario, data.hMaqIni, data.hMaqFin, dMaq, data.hProdIni, data.hProdFin, dProd, data.contIni, data.contFin, dCont, data.fesi, data.obs, data.lote, data.m3Agua, id ]; var record = { id: id, lote: data.lote, hMaq: data.hMaqFin, usuario: data.operario, hMaqIni: data.hMaqIni, hMaqFin: data.hMaqFin, hProdIni: data.hProdIni, hProdFin: data.hProdFin, contIni: data.contIni, contFin: data.contFin, fesi: data.fesi, obs: data.obs, m3Agua: data.m3Agua }; if (data.id && String(data.id).trim() !== "") { var vals = sheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (String(vals[i][16]) === id) { sheet.getRange(i+1, 1, 1, 17).setValues([fila]); return { success: true, record: record }; } } } sheet.appendRow(fila); return { success: true, record: record }; } function borrarProduccion(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(data.maquina); var vals = sheet.getDataRange().getValues(); var colId = (data.maquina === "Medios densos") ? 16 : 8; for (var i = vals.length - 1; i >= 1; i--) { if (String(vals[i][colId]).trim() === String(data.id).trim()) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, error: "No encontrado" }; } function hacerLogin(user, pass) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("config"); var datos = sheet.getDataRange().getValues(); var res = { success: false, motivos: [], operarios: [], activosFinder: [], activosZurik: [], activosMD: [], activosSteinert: [] }; for (var i = 1; i < datos.length; i++) { if (String(datos[i][0]).trim() === String(user).trim() && String(datos[i][1]).trim() === String(pass).trim()) { res.success = true; res.user = String(datos[i][0]).trim(); } if (datos[i][2]) res.motivos.push(datos[i][2]); if (datos[i][4]) res.operarios.push(datos[i][4]); if (datos[i][5]) res.activosFinder.push(datos[i][5]); if (datos[i][6]) res.activosZurik.push(datos[i][6]); if (datos[i][7]) res.activosMD.push(datos[i][7]); if (datos[i][8]) res.activosSteinert.push(datos[i][8]); } return res; } function guardarIncidencia(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("incidencias " + data.maquina); var val = validarTiempos(data.horaParada, data.horaArranque, data.turno); if (val.error) return { success: false, error: val.msg }; var id = Utilities.getUuid(); sheet.appendRow([data.fecha, data.turno, data.usuario, new Date(), data.horaParada, data.horaArranque, data.motivo, data.maquinaAfectada, data.observaciones, val.diff, id]); return { success: true, idUnico: id }; } function borrarIncidencia(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("incidencias " + data.maquina); var datos = sheet.getDataRange().getValues(); for (var i = datos.length - 1; i >= 0; i--) { if (String(datos[i][10]).trim() === String(data.idUnico).trim()) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false }; } function guardarPrueba(data) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("prueba " + data.linea.toLowerCase()); if (!sheet) return { success: false, error: "Hoja no existe" }; var val = validarTiempos(data.hInicio, data.hFin, data.turno); sheet.appendRow([data.fecha, data.usuario, new Date(), data.proveedor, data.linea, data.hInicio, data.hFin, val.diff, data.campo1, data.campo2, data.campo3, data.campo4, data.campo5, data.campo6]); return { success: true }; } function validarTiempos(inicio, fin, turno) { var p1 = String(inicio).split(":"), p2 = String(fin).split(":"); var m1 = parseInt(p1[0])*60 + parseInt(p1[1]), m2 = parseInt(p2[0])*60 + parseInt(p2[1]); if (m2 <= m1) { if (turno === "Noche" && m1 >= 1320 && m2 <= 360) m2 += 1440; else return { error: true, msg: "Revisar horas" }; } return { error: false, diff: (m2 - m1)/60 }; }
